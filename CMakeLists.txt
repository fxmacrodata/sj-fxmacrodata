cmake_minimum_required(VERSION 3.16)
project(fxmacrodata_sj LANGUAGES C CXX)
set(SJ_INCLUDE_DIR "" CACHE PATH "Directory containing reviewed original sj.h")
if(NOT EXISTS "${SJ_INCLUDE_DIR}/sj.h")
  message(FATAL_ERROR "Set SJ_INCLUDE_DIR to the independently supplied reviewed sj.h directory")
endif()
add_library(fxmacrodata_sj fxmacrodata_sj.cpp sj_impl.c)
target_compile_features(fxmacrodata_sj PUBLIC cxx_std_17)
set_target_properties(fxmacrodata_sj PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES)
target_include_directories(fxmacrodata_sj PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${SJ_INCLUDE_DIR}")
add_executable(fxmacrodata_sj_tests tests.cpp)
target_link_libraries(fxmacrodata_sj_tests PRIVATE fxmacrodata_sj)
enable_testing()
add_test(NAME fxmacrodata_sj_tests COMMAND fxmacrodata_sj_tests)
